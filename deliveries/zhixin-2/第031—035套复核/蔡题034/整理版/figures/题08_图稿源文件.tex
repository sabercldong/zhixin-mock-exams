 \pgfmathsetmacro{\ny}{#2}

  \path[fill=blue!10,fill opacity=.6,draw=blue!65]
  ({\s*\nx+\ny},{\s*\ny-\nx},-.9)
  --({\s*\nx-\ny},{\s*\ny+\nx},-.9)
  --({\s*\nx-\ny},{\s*\ny+\nx},.9)
  --({\s*\nx+\ny},{\s*\ny-\nx},.9)--cycle;
  \draw[blue!35]({\s*\nx+\ny},{\s*\ny-\nx},0)--({\s*\nx-\ny},{\s*\ny+\nx},0);
  \draw[blue!35]({\s*\nx},{\s*\ny},-.9)--({\s*\nx},{\s*\ny},.9);
 }
}
